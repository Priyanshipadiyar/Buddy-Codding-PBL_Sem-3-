// Sample Solution for Q01 (Largest Element in an Array)
// Result: PASSES ALL TEST CASES
#include <stdio.h>
#include <limits.h>

int main() {
    int n;
    if (scanf("%d", &n) != 1) return 0;
    
    int max_val = INT_MIN;
    for (int i = 0; i < n; i++) {
        int val;
        scanf("%d", &val);
        if (val > max_val) {
            max_val = val;
        }
    }
    printf("%d\n", max_val);
    return 0;
}
